```typescript
export async function GET(req: NextRequest) {
  const phaseId = req.nextUrl.searchParams.get('phaseId');
```

```typescript
export async function GET(_req: NextRequest, { params }: { params: { id: string } }) {
  const [artifact] = await db.select().from(artifactRegistry)

  if (!artifact) return NextResponse.json({ error: 'Artifact not found' }, { status: 404 });
```

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/db';
import { artifactRegistry, inputVersions, phaseInputs } from '@/db/schema';
import { eq } from 'drizzle-orm';

export async function GET(_req: NextRequest, { params }: { params: { id: string } }) {
  const [artifact] = await db.select().from(artifactRegistry)
    .where(eq(artifactRegistry.artifactId, params.id));

  if (!artifact) return NextResponse.json({ versions: [] });

  // Find all versions of this logical artifact
  const allVersions = await db.select().from(artifactRegistry)
    .where(eq(artifactRegistry.artifactName, artifact.artifactName));

  return NextResponse.json({ artifactName: artifact.artifactName, versions: allVersions });
}
```
